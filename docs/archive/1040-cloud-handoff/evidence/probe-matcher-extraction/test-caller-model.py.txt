SHA = 'a' * 40
REF = 'glm/1040-release-picker-snapshot'
PR = 1066


def run(id, status='completed', conclusion='success', updated='2026-09-23T10:19:05Z',
        created='2026-09-23T09:55:00Z', actor=BOT, trigger=BOT, sha=SHA, ref=REF,
        prs=(PR,), attempt=1, event='workflow_dispatch'):
    r = {'id': id, 'head_sha': sha, 'head_branch': ref, 'event': event,
         'actor': {'login': actor}, 'triggering_actor': {'login': trigger},
         'pull_requests': [{'number': p} for p in prs], 'run_attempt': attempt,
         'created_at': created, 'updated_at': updated}
    if status == 'completed':
        r['status'] = 'completed'
        r['conclusion'] = conclusion
    else:
        r['status'] = status
    return r


def ev(id, kind, label, created, actor='seanmccarthyns'):
    return {'id': id, 'event': kind,
            'label': ({'name': label} if label else None),
            'created_at': created, 'actor': {'login': actor}}


def rr(id, created, status='completed', sha=SHA, attempt=1):
    """A requester workflow run (event=pull_request_target). Any status counts."""
    return {'id': id, 'head_sha': sha, 'event': 'pull_request_target',
            'created_at': created, 'updated_at': created,
            'status': status, 'conclusion': 'failure' if status == 'completed' else None,
            'run_attempt': attempt}


class Api:
    """Scripted adapter. list_runs serves per-poll pages (clamped to the last); a
    dispatched run becomes discoverable only reveal_after list_runs calls AFTER the
    dispatch (delayed visibility). dispatch returns dispatch_response (None = the
    response is 'lost' — uncertain outcome)."""

    def __init__(self, runs_series=None, events=(), events_complete=True, run_records=None,
                 reveal_after=0, jobs_by_attempt=None, dispatch_response=None,
                 requester_runs=(), requester_complete=True, get_sequences=None):
        self.calls = []
        self.polls = 0
        self.series = runs_series or [[]]
        self.events = list(events)
        self.events_complete = events_complete
        self.records = dict(run_records or {})
        self.reveal_after = reveal_after
        self.jobs = dict(jobs_by_attempt or {})
        self.dispatch_response = dispatch_response
        self.requester = list(requester_runs)
        self.requester_complete = requester_complete
        self.get_sequences = dict(get_sequences or {})   # run_id -> [records..., final]
        self.last_runs = []
        self.pending = None
        self.awaiting = None
        self.since = 0

    def list_runs(self):
        self.calls.append('list_runs')
        self.polls += 1
        page = self.series[min(self.polls - 1, len(self.series) - 1)]
        if self.awaiting is not None:
            self.since += 1
            if self.since > self.reveal_after:
                page.append(self.awaiting)
                self.records[self.awaiting['id']] = self.awaiting
                self.awaiting = None
        self.last_runs = page
        return page

    def get_run(self, run_id):
        self.calls.append(f'get_run:{run_id}')
        if run_id in self.get_sequences:
            seq = self.get_sequences[run_id]
            return seq.pop(0) if len(seq) > 1 else seq[0]
        return self.records.get(run_id)

    def read_events(self):
        self.calls.append('read_events')
        return self.events, self.events_complete

    def requester_history(self):
        self.calls.append('requester_history')
        return self.requester, self.requester_complete

    def dispatch(self, inputs):
        self.calls.append(f"dispatch:{inputs['pull_request_head_sha'][:7]}:"
                          f"return_run_details={inputs.get('return_run_details')}")
        if self.pending is not None:
            self.awaiting = self.pending
            self.since = 0
        return self.dispatch_response

    def jobs_for_attempt(self, run_id, attempt):
        self.calls.append(f'jobs:{run_id}@{attempt}')
        return self.jobs.get((run_id, attempt))

RESULTS = []


def check(name, fn):
    try:
        fn()
        RESULTS.append((name, 'PASS', ''))
    except AssertionError as e:
        RESULTS.append((name, 'FAIL', str(e)))
    except Exception as e:
        RESULTS.append((name, 'ERROR', f'{type(e).__name__}: {e}'))


def expect_refusal(model, reason_prefix):
    try:
        model.run()
    except Refusal as r:
        assert r.reason.startswith(reason_prefix), f'wrong refusal: {r.reason}'
        return r
    raise AssertionError(f'expected refusal {reason_prefix}, got result {model.result}')


def dispatches(api):
    return [c for c in api.calls if c.startswith('dispatch')]

# ---- 9: timestamp ties are conservative on both sides ---------------------------------
def t9():
    SELF = rr(7001, '2026-09-23T12:30:00Z')              # the current requester, postdating E*
    # A prior requester at exactly E* consumes it.
    api = Api(events=E_FRESH, requester_runs=[rr(7000, E_NOON), SELF])
    try:
        freshness(E_FRESH, [], SHA, REF, PR, requester_runs=api.requester, self_run_id=7001)
    except Refusal as r:
        assert r.reason.startswith('REFUSE-AUTHORIZATION-CONSUMED'), r.reason
    else:
        raise AssertionError('a tie did not consume the authorization')
    # The current requester at exactly E* does not qualify for it.
    try:
        freshness(E_FRESH, [], SHA, REF, PR, requester_runs=[rr(7000, E_NOON)], self_run_id=7000)
    except Refusal as r:
        assert r.reason == 'REFUSE-REQUESTER-PREDATES-AUTHORIZATION', r.reason
    else:
        raise AssertionError('a tie let the requester qualify for the authorization')


# ---- 10: statuses all count; incomplete/unestablishable requester history refuses -----
def t10():
    SELF = rr(7001, '2026-09-23T12:30:00Z')
    # queued (pending) / running / cancelled / completed requester runs all consume.
    for status in ('pending', 'in_progress', 'cancelled', 'completed'):
        api = Api(events=E_FRESH,
                  requester_runs=[rr(7000, '2026-09-23T12:00:05Z', status=status), SELF])
        try:
            freshness(E_FRESH, [], SHA, REF, PR, requester_runs=api.requester, self_run_id=7001)
        except Refusal as r:
            assert r.reason.startswith('REFUSE-AUTHORIZATION-CONSUMED'), (status, r.reason)
        else:
            raise AssertionError(f'{status} requester run did not consume the authorization')
    # incomplete history -> refuse
    try:
        freshness(E_FRESH, [], SHA, REF, PR, requester_runs=[SELF],
                  self_run_id=7001, requester_complete=False)
    except Refusal as r:
        assert r.reason == 'REFUSE-INCOMPLETE-REQUESTER-HISTORY', r.reason
    else:
        raise AssertionError('incomplete requester history was accepted')
    # unestablishable requester timestamp -> refuse (never silently skipped)
    try:
        freshness(E_FRESH, [], SHA, REF, PR, requester_runs=[SELF, rr(7000, '')],
                  self_run_id=7001)
    except Refusal as r:
        assert r.reason == 'REFUSE-UNESTABLISHABLE-REQUESTER-HISTORY', r.reason
    else:
        raise AssertionError('a requester record with no timestamp was skipped')
    # self identity absent from the history -> refuse
    try:
        freshness(E_FRESH, [], SHA, REF, PR, requester_runs=[], self_run_id=7001)
    except Refusal as r:
        assert r.reason == 'REFUSE-SELF-IDENTITY-UNRESOLVABLE', r.reason
    else:
        raise AssertionError('a requester that cannot see itself was allowed to dispatch')

# ---- mutation self-test: a fail-open variant MUST be caught -----------------------------
def t17():
    def fail_open_freshness(events, runs, sha, ref, pr, complete=True, **_):
        boundaries = [instant(r.get('updated_at')) for r in runs
                      if r.get('status') == 'completed' and r.get('conclusion') != 'success']
        boundaries = [b for b in boundaries if b]        # silently drops invalid records
        labels = [instant(e.get('created_at')) for e in events
                  if e.get('event') == 'labeled' and (e.get('label') or {}).get('name') == 'ready-for-full-ci']
        labels = [t for t in labels if t]
        if not boundaries:
            return ('FRESH-NO-FAILURE', max(labels))
        return ('FRESH' if max(labels) > max(boundaries) else 'STALE', max(labels))

    bad = dict(F0, updated_at='invalid')

    assert v == 'FRESH-NO-FAILURE'                       # mutation reproduces the fail-open behavior

        freshness(E_REAL, [bad], SHA, REF, PR)           # the model's real helper refuses...
    except Refusal:
        return                                           # ...so an executable test catches the mutation
    raise AssertionError('strict helper unexpectedly accepted')
